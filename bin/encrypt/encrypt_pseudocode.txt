START

DECLARE base64 : STRING
DECLARE xor : STRING
DECLARE aes-256 : STRING
DECLARE blowfish : STRING
DECLARE rsa-2048 : STRING
DECLARE option : INTEGER
DECLARE tof : INTEGER

OUTPUT "Encrypt any file using one simple command!"
OUTPUT "WARNING: MUST BE ROOT TO RUN THIS COMMAND!"
OUTPUT "Types of encryption: "
OUTPUT "1. BASE64, 2. XOR, 3. AES-256"
OUTPUT "4. BLOWFISH, 5. RSA-2048"

INPUT option
IF option == 1 THEN
OUTPUT "You chose BASE64. Please go to the proper directory with the file you want to encrypt."
OUTPUT "Using BASE64 to encrypt your file. Please wait."
ELSE IF option == 2 THEN
OUTPUT "WARNING: XOR can potentially be cracked! Use at your own risk!"
OUTPUT "You chose XOR. Please go to the proper directory with the file you want to encrypt." 
ELSE IF option == 3 THEN
OUTPUT "You chose AES-256. Please go to the proper directory with the file you want to encrypt."
OUTPUT "Using AES-256 to encrypt your file. Please wait."
ELSE IF option == 4 THEN
OUTPUT "You chose BLOWFISH. Please go to the proper directory with the file you want to encrypt."
OUTPUT "Using BLOWFISH to encrypt your file. Please wait."
ELSE IF option == 5 THEN
OUTPUT "WARNING: RSA-2048 can potentially be cracked! Use at your own risk!"
OUTPUT "You chose RSA-2048. Please go to the proper directory with the file you want to encrypt."
OUTPUT "Using RSA to encrypt your file. Please wait."
ELSE
OUTPUT "Error: must choose options 1-5"
END IF

OUTPUT "Are you in the correct directory with the file you want to encrypt? (1 = yes, 2 = no)"

IF tof == 1 THEN
OUTPUT "Great, please link to the correct file by typing the full directory"
ELSE IF tof == 2 THEN
OUTPUT "Please go to the correct directory with the file that you want to encrypt"
ELSE
OUTPUT "Error: Must choose options 1 or 2"

END
